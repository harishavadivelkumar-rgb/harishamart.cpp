cmake_minimum_required(VERSION 3.25)
project(HarishaMart LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Drogon CONFIG REQUIRED)
find_package(unofficial-sqlite3 CONFIG REQUIRED)
add_executable(HarishaMart src/main.cc)
target_link_libraries(HarishaMart PRIVATE Drogon::Drogon unofficial::sqlite3::sqlite3)
target_compile_definitions(HarishaMart PRIVATE PROJECT_ROOT="${CMAKE_SOURCE_DIR}")
